# Charge le module ActiveDirectory si nécessaire
Import-Module ActiveDirectory

# 1) Création de l’OU racine ps_YnovDomain (protégée contre suppression accidentelle)
New-ADOrganizationalUnit `
    -Name "ps_YnovDomain" `
    -Path "DC=YnovDomaine,DC=com" `
    -ProtectedFromAccidentalDeletion $true

# 2) Sous ps_YnovDomain, création des OUs de premier niveau : France et International
New-ADOrganizationalUnit `
    -Name "France" `
    -Path "OU=ps_YnovDomain,DC=YnovDomaine,DC=com" `
    -ProtectedFromAccidentalDeletion $false

New-ADOrganizationalUnit `
    -Name "International" `
    -Path "OU=ps_YnovDomain,DC=YnovDomaine,DC=com" `
    -ProtectedFromAccidentalDeletion $false

# 3) Sous France : création des zones géographiques (Nord, Sud, Est, Ouest, Paris)
