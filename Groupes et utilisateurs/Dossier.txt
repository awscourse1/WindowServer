# 4) Parcourir chaque dossier et appliquer les droits via icacls
foreach ($folderName in $aclMap.Keys) {
    $fullPath = Join-Path $parentFolder $folderName

    # 4.2) Vérifier que le dossier existe
    if (-not (Test-Path $fullPath)) {
        Write-Warning "Le dossier '$fullPath' n'existe pas. Vérifiez le nom exact."
        continue
    }

    Write-Host "------ Appliquer ACL sur : $fullPath ------"

    $folderACLs = $aclMap[$folderName]

    foreach ($entry in $folderACLs.GetEnumerator()) {
        $adGroup  = $entry.Key      # ex. "YnovDomaine.com\Comptabilite"
        $permCode = $entry.Value.Trim()  # ex. "F"

        if ([string]::IsNullOrWhiteSpace($permCode)) { continue }

        # 4.5) Traduire le code en syntaxe icacls "(R)","(W)","(M)","(F)"
        $icaclsRight = switch ($permCode) {
            "R" { "(R)" }
            "W" { "(W)" }
            "M" { "(M)" }
            "F" { "(F)" }
            default {
                throw "Permission inconnue '$permCode' pour $adGroup sur $folderName"
            }
        }

        # 4.6) Appeler icacls AVEC LE GROUPE ET LA PERMISSION DANS LA MÊME CHAÎNE
        $grantArg = "$adGroup`:$icaclsRight"  
        #   → ex. "YnovDomaine.com\Comptabilite:(F)"

        $cmd = "icacls `"$fullPath`" /grant `"$grantArg`" /inheritance:r"
        Invoke-Expression $cmd

        Write-Host "  → $adGroup : $permCode"
    }

    Write-Host ""
}
