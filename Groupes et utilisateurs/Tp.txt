# 1) Charger le module ActiveDirectory
Import-Module ActiveDirectory

# 2) Définir la racine de l’arborescence AD (à adapter si besoin) :
#    OU=YnovGroup,OU=YnovDomain,DC=YnovDomaine,DC=com
$baseDN = "OU=YnovGroup,OU=YnovDomain,DC=YnovDomaine,DC=com"

# 3) Emplacement par défaut pour créer tous les comptes :
#    OU=Utilisateurs,OU=Paris-Est,OU=Paris,OU=France,$baseDN
#    (Adaptez pour chacun si vous voulez un autre OU “ville”)
$ouUtilisateursParisEst = "OU=Utilisateurs,OU=Paris-Est,OU=Paris,OU=France,$baseDN"

# 4) Mot de passe temporaire commun (à modifier en production)
$password = ConvertTo-SecureString "P@ssw0rd123" -AsPlainText -Force

# =======================================================================
# 5) CRÉATION IDÉMPOTENTE DES 11 COMPTES
#    Si le SamAccountName existe déjà, on ne le recrée pas.
# =======================================================================

# 5.1) Alice Dupont – adupont
if (-not (Get-ADUser -Filter { SamAccountName -eq "adupont" } -ErrorAction SilentlyContinue)) {
    New-ADUser `
        -Name "Alice Dupont" `
        -GivenName "Alice" `
        -Surname "Dupont" `
        -SamAccountName "adupont" `
        -UserPrincipalName "alice.dupont@ynovdomaine.com" `
        -Path $ouUtilisateursParisEst `
        -AccountPassword $password `
        -Enabled $true
    Write-Host "Création : adupont (Alice Dupont)"
} else {
    Write-Host "→ Le compte 'adupont' (Alice Dupont) existe déjà."
}

#      • Direction (Paris-Est)    : colonnes 1, 3, 10, 11  → adupont, sleroy, kmoreau, tdurand
